{-# OPTIONS --cubical --guardedness #-}
open import Agda.Primitive
open import Cubical.Induction.WellFounded
module InductOn where

private
  variable
    l l' : Level
    A D C : Set l

-- Taken from `Relation.Binary.Construct.On` from the stdlib
-- (Can't use it directly because `Acc` is the cubical one)

_on_ : (A → A → D) → (C → A) → (C → C → D)
_*_ on f = \x y -> f x * f y

module with-f (f : A → C) where
  accessible : ∀ {∼ : C → C → Set l'} {x} → Acc ∼ (f x) → Acc (∼ on f) x
  accessible (acc rs) = acc (λ y fy<fx → accessible (rs (f y) fy<fx))

  wellFounded : {∼ : C → C → Set l'} → WellFounded ∼ → WellFounded (∼ on f)
  wellFounded wf x = accessible (wf (f x))

open with-f public
