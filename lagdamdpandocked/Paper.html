  variable
    ℓ ℓC ℓC' ℓD ℓD' : Level
_^_ : {ℓ ℓC ℓC' : Level} → (C : Category ℓC ℓC') (X : Type ℓ) → Category (ℓ-max ℓ ℓC) (ℓ-max ℓ ℓC')
_^_ C A = PowerCategory A C
Endo : {ℓC ℓC' : Level} → (C : Category ℓC ℓC') → Type (ℓC ⊔ ℓC')
Endo C = Functor C C

module with-𝒞 {𝒞 : Category ℓC ℓC'} where
  open Functor

  -- lifted downset restriction
  pr : (i : A) → Functor (𝒞 ^ A) (𝒞 ^ (↓ i))
  pr i .F-ob = _∘ ↪
  pr i .F-hom = _∘ ↪
  pr i .F-id = refl
  pr i .F-seq = λ _ _ → refl

  -- evaluation (indep. of _<_)
  ev : (i : A) → Functor (𝒞 ^ A) 𝒞
  ev i .F-ob = _$ i
  ev i .F-hom = _$ i
  ev i .F-id = λ _ → 𝒞 .Category.id
  ev i .F-seq = λ f g _ → (𝒞 Category.⋆ f i) (g i)

  module with-F (F : Endo (𝒞 ^ A)) where

    isWellFounded : Type (ℓ-max (ℓ-max (ℓ-max ℓA ℓ<) ℓC) ℓC')
    isWellFounded = ∀ (i : A) → Σ[ G ∈ Functor (𝒞 ^ (↓ i)) 𝒞 ] (ev i ○ F) ≅ (G ○ pr i)

    F-wf⇒F-coalgs-recursive : (<-wf : WellFounded _<_) → (F-wf : isWellFounded) → (α : Coalgebra F) → isRecursive F α
    F-wf⇒F-coalgs-recursive <-wf F-wf (X , c) (Y , a) = (h ,  isC2AMorph-h) ,  unique where
      open WFI <-wf
      open NatTrans
      open Category using() renaming (Hom[_,_] to Hom[_][_,_])
      open Category 𝒞

      F< : (i : A) → Functor (𝒞 ^ (↓ i)) 𝒞
      F< i = F-wf i .fst

      iso : (i : A) → (ev i ○ F) ≅ ((F< i) ○ pr i)
      iso i = F-wf i .snd

      ϕ : (i : A) → (Z : Category.ob (𝒞 ^ A)) → Hom[ (ev i ○ F) .F-ob Z , ((F< i) ○ pr i) .F-ob Z ]
      ϕ i Z = iso i ._≅_.trans .N-ob Z

      ϕ⁻¹ : (i : A) → (Z : Category.ob (𝒞 ^ A)) → Hom[ ((F< i) ○ pr i) .F-ob Z , (ev i ○ F) .F-ob Z ]
      ϕ⁻¹ i Z = (iso i ._≅_.nIso Z) .isIso.inv

      g : (i : A) → (IH : (j : A) → j < i → Hom[ 𝒞 ][ X j , Y j ]) →
         Hom[ 𝒞 ^ (↓ i) ][ pr i .F-ob X , pr i .F-ob Y ]
      g _ = uncurry

      step : (i : A) → ((j : A) → j < i → Hom[ 𝒞 ][ X j , Y j ]) → Hom[ X i , Y i ]
      step i IH =
        c i ⋆ (ϕ i X ⋆ (F< i) .F-hom (g i IH) ⋆ ϕ⁻¹ i Y) ⋆ a i

      h : Hom[ 𝒞 ^ A ][ X , Y ]
      h = induction step

      preq : (i : A) → pr i .F-hom h ≡ g i (λ j _ → h j)
      preq i = funExt (λ {(j , j<i) → refl })

      ϕϕ⁻¹≡id : (i : A) → (Z : Category.ob (𝒞 ^ A)) → ϕ i Z ⋆ ϕ⁻¹ i Z ≡ id
      ϕϕ⁻¹≡id i Z = iso i ._≅_.nIso Z .isIso.ret

      isC2AMorph-hᵢ : (i : A) → h i ≡ c i ⋆ F .F-hom h i ⋆ a i
      isC2AMorph-hᵢ i = h i ≡⟨ induction-compute step i ⟩
          c i ⋆ (ϕ i X ⋆ (F< i) .F-hom (g i (λ j _ → h j)) ⋆ ϕ⁻¹ i Y) ⋆ a i ≡⟨
          cong (λ o → c i ⋆ o ⋆ a i) (
           ϕ i X ⋆ (F< i) .F-hom (g i (λ j _ → h j)) ⋆ ϕ⁻¹ i Y ≡⟨
             cong (λ o → ϕ i X ⋆ (o ⋆ ϕ⁻¹ i Y)) (
               (F< i) .F-hom (g i (λ j _ → h j)) ≡⟨ cong ((F< i) .F-hom) (sym (preq i)) ⟩
               ((F< i) ○ pr i) .F-hom h ≡⟨ _≅_.sqLR (iso i) ⟩
                 ((ϕ⁻¹ i X ⋆ (ev i ○ F) .F-hom h) ⋆ ϕ i Y) ∎
                 )
             ⟩
           ϕ i X ⋆ ((ϕ⁻¹ i X ⋆ F .F-hom h i) ⋆ ϕ i Y) ⋆ ϕ⁻¹ i Y ≡⟨  solveCat! 𝒞  ⟩
           (ϕ i X ⋆ ϕ⁻¹ i X) ⋆ F .F-hom h i ⋆ (ϕ i Y ⋆ ϕ⁻¹ i Y) ≡⟨
             cong₂ (λ l r → l ⋆ F .F-hom h i ⋆ r) (ϕϕ⁻¹≡id i X) (ϕϕ⁻¹≡id i Y) ⟩
           id ⋆ F .F-hom h i ⋆ id ≡⟨ ⋆IdL _ ∙ ⋆IdR _ ⟩
           F .F-hom h i ∎
           )⟩
          (c i ⋆ F .F-hom h i ⋆ a i) ∎

      isC2AMorph-h : isC2AMorph F c a h
      isC2AMorph-h = funExt isC2AMorph-hᵢ


      ≡-step : (u : Hom[ 𝒞 ^ A ][ X , Y ]) → (isC2AMorph-u : isC2AMorph F c a u) →
        (i : A) → ((j : A) → j < i → h j ≡ u j) → h i ≡ u i
      ≡-step u isC2AMorph-u i IH =
        h i ≡⟨ isC2AMorph-hᵢ i ⟩
        c i ⋆ F .F-hom h i ⋆ a i ≡⟨
          cong (λ o → c i ⋆ (o ⋆ a i)) (
            F .F-hom h i ≡⟨ _≅_.sqRL (iso i) ⟩
            (ϕ i X ⋆ ((F< i) ○ pr i) .F-hom h) ⋆ ϕ⁻¹ i Y ≡⟨
              cong (λ o → (ϕ i X ⋆ F< i .F-hom o) ⋆ ϕ⁻¹ i Y)
              aux ⟩
            (ϕ i X ⋆ ((F< i) ○ pr i) .F-hom u) ⋆ ϕ⁻¹ i Y ≡⟨ sym (_≅_.sqRL (iso i)) ⟩
            (ev i ○ F) .F-hom u ∎
           ) ⟩
        ( c i ⋆ F .F-hom u i ⋆ a i) ≡⟨ funExt⁻ (sym isC2AMorph-u) i ⟩
        u i ∎ where
        aux : pr i .F-hom h ≡ pr i .F-hom u
        aux = funExt (λ {(j , j<i) → IH j j<i})

      unique : (u : C2AMorph F (X , c) (Y , a)) → (h , isC2AMorph-h) ≡ u
      unique (u , isC2AMorph-u) = let
         h≡u : h ≡ u
         h≡u = funExt $ induction $ ≡-step u isC2AMorph-u
         in ΣPathP (h≡u , isProp→PathP (λ i → isPropIsC2AMorph F c a (h≡u i)) isC2AMorph-h isC2AMorph-u )

    open import Cubical.Categories.Limits.Initial
    module with-⊥ (⊥ : Initial 𝒞) (i : A) where
      open Category

      em : Functor (𝒞 ^ (↓ i)) (𝒞 ^ (Σ[ j ∈ A ] (Dec (j < i))))
      em = record
        { F-ob  = emF-ob
        ; F-hom = emF-hom
        ; F-id  = funExt λ {(j , yes j<i) → refl ; (j , no j≮i) → refl}
        ; F-seq = λ _ _ → funExt λ {(j , yes j<i) → refl ; (j , no j≮i) → sym (𝒞 .⋆IdR (𝒞 .id ))}
        } where
        emF-ob : (𝒞 ^ (↓ i)) .ob → (𝒞 ^ (Σ[ j ∈ A ] (Dec (j < i)))) .ob
        emF-ob x (j , yes j<i) = x (j , j<i)
        emF-ob x (j , no ¬j<i) = ⊥ .fst

        emF-hom : {x y : ob (𝒞 ^ (↓ i))} → (𝒞 ^ (↓ i)) [ x , y ] →
          (𝒞 ^ (Σ[ j ∈ A ]  Dec (j < i))) [ emF-ob x , emF-ob y ]
        emF-hom x (j , yes j<i) = x (j , j<i)
        emF-hom x (j , no ¬j<i) = 𝒞 .id

      em2 : Functor (𝒞 ^ (↓ i)) (ΠC A (λ j → 𝒞 ^ (Dec (j < i))))
      em2 = record
        { F-ob  = em2F-ob
        ; F-hom = em2F-hom
        ; F-id  = funExt (λ j → funExt (λ{(yes j<i) → refl ; (no j≮i) → refl}))
        ; F-seq = λ _ _ → funExt λ j → funExt λ{(yes j<i) → refl ; (no j≮i) → sym (𝒞 .⋆IdR (𝒞 .id ))}
        } where
        em2F-ob : (𝒞 ^ (↓ i)) .ob → (ΠC A (λ j → 𝒞 ^ (Dec (j < i)))) .ob
        em2F-ob x j = λ{(yes j<i) → x (j , j<i); (no ¬j<i) → ⊥ .fst}

        em2F-hom : {x y : ob (𝒞 ^ (↓ i))} → (𝒞 ^ (↓ i)) [ x , y ] →
          (ΠC A (λ j → 𝒞 ^ (Dec (j < i)))) [ em2F-ob x , em2F-ob y ]
        em2F-hom x j (yes j<i) = x (j , j<i)
        em2F-hom x j (no ¬j<i) = 𝒞 .id

      module with-<? (_<?_ : (i j : A) → Dec (i < j)) where
        -- with decidability (note: This is _not_ how we do it in the library, all the applications rely on providing proofs manually)
        incl : Functor (𝒞 ^ (↓ i)) (𝒞 ^ A)
        incl .F-ob x j = em2 .F-ob x j (j <? i)
        incl .F-hom x a = em2 .F-hom x a (a <? i)
        incl .F-id = funExt (λ j → funExt⁻ (funExt⁻ (em2 .F-id) j) (j <? i))
        incl .F-seq = λ f g → funExt λ j → funExt⁻ (funExt⁻ (em2 .F-seq f g) j) (j <? i)
      open with-<? public

    open with-⊥ public
  open with-F public
