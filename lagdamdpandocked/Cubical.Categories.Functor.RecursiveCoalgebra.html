{-# OPTIONS --cubical --guardedness #-}
{- Recursive Coalgebras for a functor. Adapted from
https://arxiv.org/src/2405.09504/anc/Coalgebra.Recursive.html
as well as ./(Co)algebra,
and cubical/Cubical.Categories.Limits.Initial
-}
module Cubical.Categories.Functor.RecursiveCoalgebra where

open import Cubical.Foundations.Prelude
open import Cubical.Foundations.HLevels

open import Cubical.Data.Sigma

open import Cubical.Categories.Category
open import Cubical.Categories.Functor

open import Cubical.Categories.Functor.Algebra
open import Cubical.Categories.Functor.Coalgebra


private
  variable
    ℓ ℓ' : Level

module _ {C : Category ℓ ℓ'} (F : Functor C C) where
  open Category
  open Category C using () renaming (_⋆_ to _⨾_)
  open Functor F

  isC2AMorph : ∀ {a b} → CoalgebraFor F a → AlgebraFor F b → (C [ a , b ]) → Type _
  isC2AMorph α κ ϕ =  ϕ ≡ α ⨾ F-hom ϕ ⨾ κ

  isPropIsC2AMorph : ∀ {a b} →
                       (α : CoalgebraFor F a)(β : AlgebraFor F b)
                       (ϕ : C [ a , b ]) → isProp (isC2AMorph α β ϕ)
  isPropIsC2AMorph α β ϕ = (C .isSetHom) _ _

  C2AMorph : Coalgebra F → Algebra F → Type ℓ'
  C2AMorph α β =
    Σ[ ϕ ∈ C [ α .fst , β .fst ] ] isC2AMorph (α .snd) (β .snd) ϕ

  isSetC2AMorphs : ∀ {α β} → isSet (C2AMorph α β)
  isSetC2AMorphs {x}{y} =
    isSetΣ (C .isSetHom)
      (λ ϕ → isProp→isSet (isPropIsC2AMorph (x .snd) (y .snd) ϕ))

  isRecursive : Coalgebra F → Type (ℓ-max ℓ ℓ')
  isRecursive α = ∀ (κ : Algebra F) → isContr (C2AMorph α κ)

  isPropIsRecursive : (x : Coalgebra F) → isProp (isRecursive x)
  isPropIsRecursive _ = isPropΠ λ _ → isPropIsContr
