module _ {a b : A} {xs : ℬ A} where

  ≤-to-# : a ≤ b → All (b ≤h_) xs → All (a ≤h_) xs

-- This is @bentnib's `permBubble` and is used whenever we have inserted an element in the right subtree, but need to find it at the top for bookkeeping purposes

→↑ : {x : B} → (iₗ : ℬ B) → { iᵣ : ℬ B} →
  iₗ ++ (x ∷  iᵣ) ≡ x ∷ iₗ ++  iᵣ
→↑ {x = x} iₗ { iᵣ} =
   iₗ ++ (x ∷  iᵣ) ≡⟨ (comm-++ iₗ (x ∷  iᵣ))⟩

   x ∷ iₗ ++  iᵣ ∎

module _ {l : Level} {B : Type l} where

  data L (r : ℬ B → Type l') : ℬ B → Type (l ⊔ l') where
    []   : L r []
    _∷_  : {g : ℬ B} (x : B) → (r g) → L r (x ∷ g)

  data EIList : ℬ B → Type l where
    []  : EIList []
    _∷_ : {g : ℬ B} (x : B) → (EIList g) → EIList (x ∷ g)

  inL : (L EIList) ⇒ EIList
  inL _ [] = []
  inL _ (x ∷ xs) = x ∷ xs

paraL : {l : Level} → {A : Type l} → {r : ℬ A → Type l} →
  (L (EIList ⊠ r) ⇒ r) → EIList ⇒ r
paraL alg _ [] = alg _ []
paraL alg _ (x ∷ l) = alg _ (x ∷ (l , paraL alg _ l))


data O (r : ℬ A → Type) : ℬ A → Type where
  []    : O r []
  _≤∷_  : {g : ℬ A} (x : A) → (rg : r g) →
     All (x ≤h_) g → O r (x ∷ g)

data OEIList : ℬ A → Type where
  []   : OEIList []
  _≤∷_ :  {g : ℬ A} (x : A) → (rg : OEIList g) →
     All (x ≤h_) g → OEIList (x ∷ g)

outO : OEIList ⇒ (O OEIList)
outO _ [] = []
outO _ ((x ≤∷ xs) prf) = (x ≤∷ xs) prf

inO : (O OEIList) ⇒ OEIList
inO i [] = []
inO i ((x ≤∷ rg) x₁) = (x ≤∷ rg) x₁

O₁ : {r r′ : ℬ A → Type} → (r ⇒ r′) → O r ⇒ O r′
O₁ f _ [] = []
O₁ f _ ((x ≤∷ rg) prf) = (x ≤∷ f _ rg) prf

pattern _≤∷_^_bc_ x rg g prf = _≤∷_ {g = g} x rg prf

      shuffle1 : ∀ {iₗ  iᵣ : ℬ A} {x : A} → length iₗ + suc (length  iᵣ) ≡ length (x ∷ iₗ ++  iᵣ)
      shuffle1 {iₗ} { iᵣ} {x} =

        length iₗ + length (x ∷  iᵣ) ≡⟨ sym length-++ ⟩

ε-split-epi : (i : ℬ A) → {r : ℬ A → Type} → (x : S r i) → S₁ sndᵢ i (S-ε⁻¹ i x) ≡ x
ε-split-epi _ leaf = refl
ε-split-epi _ ((tₗ ┌ x ┐ tᵣ) x₁ x₂) = refl

S-hom : {X Y Z : ℬ A → Type} →
  (f : (Y ⇒ Z)) → (g : (X ⇒ Y)) →
  (i : ℬ A) → S₁ (f ∘ᵢ g) i ≡ (S₁ f ∘ᵢ S₁ g) i
S-hom {X = X} f g i = funExt biz where
  biz : (x : S X i) → (S₁ (f ∘ᵢ g) i x) ≡ ((S₁ f ∘ᵢ S₁ g) i x)
  biz leaf = refl
  biz ((tₗ ┌ x ┐ tᵣ) x₁ x₂) = refl

nat-sq : {r r' : ℬ A → Type} → (h : r ⇒ r') → (i : ℬ A) →
       S₁ h i ≡ S₁ sndᵢ i ∘ (guard₀₁ S₁ h) i ∘ S-ε⁻¹ i
nat-sq {r = r} h i = funExt biz where
  biz : (x : S r i) → (S₁ h i x) ≡ ((S₁ sndᵢ i ∘ guard₀₁ S₁ h i ∘ S-ε⁻¹ i) x)
  biz leaf = refl
  biz ((tₗ ┌ x ┐ tᵣ) x₁ x₂) = refl

recCoalgMorphS : {r : ℬ A → Type} (c : r ⇒ S r)
                 {X : ℬ A → Type} (a : S X ⇒ X) →
                  r ⇒ X
recCoalgMorphS c a = c2a-mor.c2a-morph c a where
    module c2a-mor = U.c2a <l-wellfounded S S₁ S-ε⁻¹
    module c2a-hom = c2a-mor.c2a c a S-hom ε-split-epi
    module unique = c2a-hom.unique nat-sq

QuickSortFunctorWellfounded : is-Wellfounded S
QuickSortFunctorWellfounded i = ([ S ]< i) , (λ X → iso (S-ε⁻¹ i) (S₁ sndᵢ i) (ε-split-mono i) (ε-split-epi i)) where
  ε-split-mono : {X : ℬ A → Type} (j : ℬ A) → section (S-ε⁻¹ {X = X} j) (S₁ sndᵢ j)
  ε-split-mono _ leaf = refl
  ε-split-mono _ (((tₗ , tₗ<o) ^ iₗ ┌ x ┐ (tᵣ , tᵣ<o) ^  iᵣ) pf₁ pf₂) =
    λ i₁ → ((<l-Prop _ _ (i<x∷i++ iᵣ) tₗ i₁ , tₗ<o) ┌ x ┐ (<l-Prop _ _ i<x∷[ iₗ ]++i tᵣ i₁ , tᵣ<o)) pf₁ pf₂


sprout : {r : ℬ A → Type} → L ((S ₓ) r) ⇒ S ((L ₊) r)
sprout _ []                                  = leaf
sprout _ (x ∷ (fst₁ , leaf))                 = (inl fst₁ ┌ x ┐ inl fst₁) []-A []-A
sprout _ (x ∷ (fst₁ , (lt ┌ x₁ ┐ rt) x₂ x₃)) with x ≤?≥ x₁
...| inl x≤x₁ = (inr (x ∷ lt) ┌ x₁ ┐ inl rt) (x≤x₁ ∷-A x₂) x₃ €
       subst (S ((L ₊) _)) comm⇃
sprout _ (x ∷ (fst₁ , (lt ^ iₗ ┌ x₁ ┐ rt ^ _) x₂ x₃)) |
     inr x₁≤x = (inl lt ┌ x₁ ┐ inr (x ∷ rt)) x₂ (x₁≤x ∷-A x₃) €
       subst (S ((L ₊) _)) $ →↑ (_ ∷ iₗ)

wither : {r : ℬ A → Type} → S ((O ₓ) r) ⇒ O ((S ₊) r)
wither _ leaf = []
wither _ (((fst₁ , []) ┌ x ┐ (fst₂ , snd₂)) x₁ x₂) = (x ≤∷ inl fst₂) x₂
wither _ (((fst₁ , (x₃ ≤∷ rg) x₄) ┌ x ┐ (fst₂ , snd₂)) x₁ x₂) with uncons-A x₁
... | (x₃≤x , All-x≤-g) = (x₃ ≤∷ inr ((rg ┌ x ┐ fst₂) All-x≤-g x₂))
  (x₃≤x ∷-A x₄ ++-A ≤-to-# x₃≤x x₂) €
  subst (O ((S ₊) _)) comm⇃

quickSort : EIList ⇒ OEIList
quickSort = recCoalgMorphS (paraL (S₁ [ idᵢ ⸴ inL ] ∘ᵢ sprout)) (apoO (wither ∘ᵢ S₁ (⟨ idᵢ ⸴ outO ⟩ )))
