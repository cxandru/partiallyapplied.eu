-- constructing all decompositions of
-- (non-empty) lists into smaller (non-empty) lists
-- 
module ListNonEmptyDecompositions {a : Level} {A : Set a} where
  ⁺∷-inj₁ : ∀ {x y : A} {xs ys : List A} → (x List⁺.∷ xs ≡ y List⁺.∷ ys) → x ≡ y
  ⁺∷-inj₁ = cong (λ l → proj₁ (List⁺.uncons l))
  ⁺∷-inj₂ : ∀ {x y : A} {xs ys : List A} → (x List⁺.∷ xs ≡ y List⁺.∷ ys) → xs ≡ ys
  ⁺∷-inj₂ = cong (λ l → proj₂ (List⁺.uncons l))
  []≠∷ : ∀ {x : A} {xs : List A} → ([] ≡ x List.∷ xs) → ⊥
  []≠∷ {x} {xs} ()

  record Decomposition  (w : List⁺ A) : Set a where
    constructor _+d_≡d_
    field
      u : List⁺ A
      v : List⁺ A
      u++v≡w : u ⁺++⁺ v ≡ w
  
    u,v : List⁺ A × List⁺ A
    u,v = u , v

    to-∃₂ : ∃[ x ] ∃[ y ] u ⁺++⁺ v ≡ w
    to-∃₂ = u , v , u++v≡w

  -- finding all decompositions of a non-empty list (a ∷ w) into
  -- two non-empty lists:
  module without-annotations where
    decompositions : A → (w : List A)
                     → List (List⁺ A × List⁺ A)
    decompositions _ [] = []
    decompositions x (y ∷ ys) =
      (List⁺.[ x ] , y ∷ ys)
      ∷
      List.map (λ { (u ∷ us , v) → x ∷ u ∷ us , v}) (decompositions y ys)

    decompositions⁺ : ∀ {x : A} {w : List A} (d : Decomposition ( x ∷ w ))
                      → Decomposition.u,v d ∈ decompositions x w
    decompositions⁺ {x} {[]} ((u ∷ []) +d v ≡d ())
    decompositions⁺ {x} {[]} ((u ∷ u' ∷ us) +d v ≡d ())
    decompositions⁺ {x} {w ∷ ws} ((u ∷ []) +d v ≡d u++v≡w) with (⁺∷-inj₁ u++v≡w)
    decompositions⁺ {x} {w ∷ ws} ((u ∷ []) +d (v ∷ v') ≡d u++v≡w) | refl =
      here (cong₂ _,_
               (cong (_∷ []) (⁺∷-inj₁ u++v≡w))
               (cong₂ List⁺._∷_ (∷-injectiveˡ (⁺∷-inj₂ u++v≡w))
                                (∷-injectiveʳ (⁺∷-inj₂ u++v≡w))))
    decompositions⁺ {x} {w ∷ ws} ((u ∷ u' ∷ us) +d v ≡d uu'us++v≡w) with (⁺∷-inj₁ uu'us++v≡w)
    decompositions⁺ {x} {w ∷ ws} ((u ∷ u' ∷ us) +d (v ∷ vs) ≡d uu'us++v≡w) | refl =
      there (∈-map⁺ _ decom-w)
      where
        u'us++v≡w : u' ∷ us ++ v ∷ vs ≡ w List.∷ ws
        u'us++v≡w = ⁺∷-inj₂ uu'us++v≡w
        u'us++v≡⁺w : u' ∷ us ++ v ∷ vs ≡ w List⁺.∷ ws
        u'us++v≡⁺w = cong₂ List⁺._∷_ (∷-injectiveˡ u'us++v≡w) (∷-injectiveʳ u'us++v≡w)
        decom-w : ((u' ∷ us) , v ∷ vs) ∈ decompositions w ws
        decom-w = decompositions⁺ {w} {ws} ((u' ∷ us) +d (v ∷ vs) ≡d u'us++v≡⁺w)

    decompositions⁻ : (w₀ : A) → (w : List A)
                     → ∀ uv → (uv ∈ decompositions w₀ w) → (proj₁ uv ⁺++⁺ proj₂ uv ≡ w₀ ∷ w)
    decompositions⁻ x [] = λ uv ()
    decompositions⁻ x (y ∷ ys) uv (here refl) = refl
    decompositions⁻ x (y ∷ ys) (x' ∷ [] , v) (there elem) =
      let
        ((s ∷ t ) , v') , x'v'∈decomp-y-ys , eq = (∈-map⁻ _ elem)
        eq1  : [] ≡ s ∷ t
        eq1  = ⁺∷-inj₂ (×.,-injectiveˡ eq)
      in
      ⊥-elim ([]≠∷ eq1)
    decompositions⁻ x (y ∷ ys) (x' ∷ (s ∷ t) , v) (there elem) =
      let
        ((s' ∷ t' ) , v') , x'v'∈decomp-y-ys , eq = (∈-map⁻ _ elem)

        x'≡x : x' ≡ x
        x'≡x = ⁺∷-inj₁ (×.,-injectiveˡ eq)

        v≡v' : v ≡ v'
        v≡v' = ×.,-injectiveʳ eq

        s't'v'≡yys : (s' ∷ t') ⁺++⁺ v' ≡ y ∷ ys
        s't'v'≡yys = decompositions⁻ y ys ((s' ∷ t') , v') x'v'∈decomp-y-ys

        eq1  : (s ∷ t) ≡ s' ∷ t'
        eq1  = ⁺∷-inj₂ (×.,-injectiveˡ eq)
        s≡s' : s ≡ s'
        s≡s'  = ∷-injectiveˡ eq1
        t≡t' : t ≡ t'
        t≡t'  = ∷-injectiveʳ eq1

        eq2 : (s ∷ t) ⁺++⁺ v ≡ (s' ∷ t') ⁺++⁺ v'
        eq2 = cong₂ _⁺++⁺_ (cong₂ _∷_ s≡s' t≡t') v≡v'
      in
      cong₂ (λ L R → L ∷ R) x'≡x (
        begin
        (s ∷ t) ++ v .List⁺.head ∷ v .List⁺.tail
          ≡⟨ List⁺.toList-⁺++⁺ (s ∷ t) v ⟩
        toList ((s ∷ t) ⁺++⁺ v )
          ≡⟨ cong toList eq2 ⟩
        toList ((s' ∷ t') ⁺++⁺ v' )
          ≡⟨ cong toList s't'v'≡yys ⟩
        y ∷ ys
        ∎
        )
      where
        open ≡.≡-Reasoning

  all-decompositions : ∀ (w : List⁺ A) → AllInhabitants (λ { (u , v) → u ⁺++⁺ v ≡ w})
  all-decompositions (w₁ ∷ w₂) = record
    { witnesses = decompositions w₁ w₂
    ; correct =
        (λ { {u , v} uv≡w → decompositions⁺ (u +d v ≡d uv≡w)})
        ,
        λ { {u , v} uv∈decomp → decompositions⁻ w₁ w₂ (u , v) uv∈decomp }
    }
    where
      open without-annotations
        

  -- decomposition with annotations
  decompositions' : (w₀ : A) → (w : List A)
                   → List (∃[ u ] ∃[ v ] u ⁺++⁺ v ≡ w₀ ∷ w)
  decompositions' _ [] = []
  decompositions' x (y ∷ ys) =
    (List⁺.[ x ] , y ∷ ys , refl)
    ∷
    List.map (λ { (u ∷ us , v , uv≡w) →
                  x ∷ u ∷ us , v , cong (λ { (h ∷ t) → x ∷ h ∷ t}) uv≡w})
             (decompositions' y ys)
