{-# OPTIONS --cubical --guardedness #-}
{- Algebras for a functor. Adapted from
cubical-categorical-logic/Cubical/Categories/Monad/Algebra.agda
and
agda-categories/Categories/Category/Construction/F-Algebras.agda
-}
module Cubical.Categories.Functor.Algebra where

open import Cubical.Foundations.Prelude
open import Cubical.Foundations.HLevels

open import Cubical.Data.Sigma

open import Cubical.Categories.Category
open import Cubical.Categories.Functor

private
  variable
    ℓ ℓ' : Level

module _ {C : Category ℓ ℓ'} (F : Functor C C) where
  open Category
  open Category C using () renaming (_⋆_ to _⨾_)
  open Functor F

  AlgebraFor : (a : C .ob) → Type ℓ'
  AlgebraFor a = C [ F-ob a , a ]

  isAlgebraHom : ∀ {a b} → AlgebraFor a → AlgebraFor b → (C [ a , b ]) → Type _
  isAlgebraHom α β ϕ =  α ⨾ ϕ ≡ F-hom ϕ ⨾ β

  isPropIsAlgebraHom : ∀ {a b} →
                       (α : AlgebraFor a)(β : AlgebraFor b)
                       (ϕ : C [ a , b ]) → isProp (isAlgebraHom α β ϕ)
  isPropIsAlgebraHom α β ϕ = (C .isSetHom) _ _

  Algebra = Σ[ a ∈ C .ob ] AlgebraFor a

  AlgebraHom : Algebra → Algebra → Type _
  AlgebraHom α β =
    Σ[ ϕ ∈ C [ α .fst , β .fst ] ] isAlgebraHom (α .snd) (β .snd) ϕ

  isSetAlgebraHom : ∀ {α β} → isSet (AlgebraHom α β)
  isSetAlgebraHom {x}{y} =
    isSetΣ (C .isSetHom)
      (λ ϕ → isProp→isSet (isPropIsAlgebraHom (x .snd) (y .snd) ϕ))

  AlgebraHom≡ : ∀ {α β} (ϕ ψ : AlgebraHom α β) → ϕ .fst ≡ ψ .fst → ϕ ≡ ψ
  AlgebraHom≡ {α}{β} ϕ ψ p =
    Σ≡Prop (λ ϕ → isPropIsAlgebraHom (α .snd) (β .snd) ϕ) p

  ALGEBRA : Category _ _
  ALGEBRA .ob = Algebra
  ALGEBRA .Hom[_,_] = AlgebraHom
  ALGEBRA .id {x} =
    C .id , C .⋆IdR _ ∙ sym (C .⋆IdL _) ∙ cong ( _⨾ (x .snd)) (sym F-id)
  ALGEBRA ._⋆_ {z = γ} ϕ ψ = 
    (ϕ .fst ⋆⟨ C ⟩ ψ .fst) ,
     sym (C .⋆Assoc _ _ _) ∙
     cong (_⨾ (ψ .fst)) (ϕ .snd)  ∙
     (C .⋆Assoc _ _ _) ∙
     cong (_⨾_ (F-hom (ϕ .fst))) (ψ .snd) ∙
     sym (C .⋆Assoc _ _ _) ∙
     cong (_⨾ _) (sym (F-seq (ϕ .fst) (ψ .fst)))
  ALGEBRA .⋆IdL {x}{y} ϕ =
    Σ≡Prop (isPropIsAlgebraHom (x .snd) (y .snd)) (C .⋆IdL _)
  ALGEBRA .⋆IdR {x}{y} ϕ =
    Σ≡Prop (isPropIsAlgebraHom (x .snd) (y .snd)) (C .⋆IdR _)
  ALGEBRA .⋆Assoc {w}{x}{y}{z} f g h =
    Σ≡Prop (isPropIsAlgebraHom (w .snd) (z .snd)) (C .⋆Assoc _ _ _)
  ALGEBRA .isSetHom {x}{y} =
    isSetΣ (C .isSetHom)
      (λ ϕ → isProp→isSet (isPropIsAlgebraHom (x .snd) (y .snd) ϕ))
