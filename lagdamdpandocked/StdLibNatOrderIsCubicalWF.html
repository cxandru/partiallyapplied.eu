open import Cubical.Relation.Nullary.Base using (¬_)
open import Cubical.Data.Sum as ⊎
module StdLibNatOrderIsCubicalWF where

private
  caseNat : ∀ {ℓ} → {A : Type ℓ} → (a0 aS : A) → ℕ → A
  caseNat a0 aS zero    = a0
  caseNat a0 aS (suc n) = aS


  snotz : {n : ℕ} → ¬ (suc n ≡ 0)
  snotz eq = subst (caseNat ⊥ ℕ) eq 0

  ¬-<-zero : {m : ℕ} → ¬ m < 0
  ¬-<-zero ()

  ≤0→≡0 : {n : ℕ} → n ≤ 0 → n ≡ 0
  ≤0→≡0 {zero} ineq = refl
  ≤0→≡0 {suc n} ineq = ⊥.rec (¬-<-zero ineq)

  m<1⇒m≡0 : {m : ℕ} → m < 1 → m ≡ 0
  m<1⇒m≡0 x = ≤0→≡0 (≤-pred x)

  <-split : {m n : ℕ} → m < suc n → (m < n) ⊎ (m ≡ n)
  <-split {n = zero} x = inr (m<1⇒m≡0 x)
  <-split {zero} {suc n} = λ _ → inl (s≤s z≤n)
  <-split {suc m} {suc n} = ⊎.map s≤s (cong suc) ∘ <-split ∘ ≤-pred

  acc-suc : {n : ℕ} → Acc _<_ n → Acc _<_ (suc n)
  acc-suc a
    = acc λ y y<sn
    → case <-split y<sn of λ
    { (inl y<n) → access a y y<n
    ; (inr y≡n) → subst _ (sym y≡n) a
    }

<-wellFounded : WellFounded _<_
<-wellFounded zero = acc (λ y →  ⊥.rec ∘ ¬-<-zero)
<-wellFounded (suc n) = acc-suc (<-wellFounded n)
