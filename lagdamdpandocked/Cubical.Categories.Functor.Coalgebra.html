{-# OPTIONS --cubical --guardedness #-}
{- Coalgebras for a functor. "Dualized" from Algebra
-}
module Cubical.Categories.Functor.Coalgebra where

open import Cubical.Foundations.Prelude
open import Cubical.Foundations.HLevels

open import Cubical.Data.Sigma

open import Cubical.Categories.Category
open import Cubical.Categories.Functor

private
  variable
    ℓ ℓ' : Level

module _ {C : Category ℓ ℓ'} (F : Functor C C) where
  open Category
  open Category C using () renaming (_⋆_ to _⨾_)
  open Functor F

  CoalgebraFor : (a : C .ob) → Type ℓ'
  CoalgebraFor a = C [ a , F-ob a ]

  isCoalgebraHom : ∀ {a b} → CoalgebraFor a → CoalgebraFor b → (C [ a , b ]) → Type _
  isCoalgebraHom α β ϕ =  α ⨾ F-hom ϕ ≡ ϕ ⨾ β

  isPropIsCoalgebraHom : ∀ {a b} →
                       (α : CoalgebraFor a)(β : CoalgebraFor b)
                       (ϕ : C [ a , b ]) → isProp (isCoalgebraHom α β ϕ)
  isPropIsCoalgebraHom α β ϕ = (C .isSetHom) _ _

  Coalgebra = Σ[ a ∈ C .ob ] CoalgebraFor a

  CoalgebraHom : Coalgebra → Coalgebra → Type _
  CoalgebraHom α β =
    Σ[ ϕ ∈ C [ α .fst , β .fst ] ] isCoalgebraHom (α .snd) (β .snd) ϕ

  isSetCoalgebraHom : ∀ {α β} → isSet (CoalgebraHom α β)
  isSetCoalgebraHom {x}{y} =
    isSetΣ (C .isSetHom)
      (λ ϕ → isProp→isSet (isPropIsCoalgebraHom (x .snd) (y .snd) ϕ))

  CoalgebraHom≡ : ∀ {α β} (ϕ ψ : CoalgebraHom α β) → ϕ .fst ≡ ψ .fst → ϕ ≡ ψ
  CoalgebraHom≡ {α}{β} ϕ ψ p =
    Σ≡Prop (λ ϕ → isPropIsCoalgebraHom (α .snd) (β .snd) ϕ) p

  COALGEBRA : Category _ _
  COALGEBRA .ob = Coalgebra
  COALGEBRA .Hom[_,_] = CoalgebraHom
  COALGEBRA .id {x} =
    C .id , cong ( _⨾_ (x .snd)) F-id ∙ C .⋆IdR _ ∙ sym (C .⋆IdL _)
  COALGEBRA ._⋆_ {z = γ} ϕ ψ =
    (ϕ .fst ⋆⟨ C ⟩ ψ .fst) ,
      cong (_⨾_ _) (F-seq (ϕ .fst) (ψ .fst)) ∙
      sym (C .⋆Assoc _ _ _) ∙
      cong (_⨾ (F-hom (ψ .fst))) (ϕ .snd) ∙
      (C .⋆Assoc _ _ _) ∙ cong (_⨾_ (ϕ .fst)) (ψ .snd) ∙
      sym (C .⋆Assoc _ _ _)
  COALGEBRA .⋆IdL {x}{y} ϕ =
    Σ≡Prop (isPropIsCoalgebraHom (x .snd) (y .snd)) (C .⋆IdL _)
  COALGEBRA .⋆IdR {x}{y} ϕ =
    Σ≡Prop (isPropIsCoalgebraHom (x .snd) (y .snd)) (C .⋆IdR _)
  COALGEBRA .⋆Assoc {w}{x}{y}{z} f g h =
    Σ≡Prop (isPropIsCoalgebraHom (w .snd) (z .snd)) (C .⋆Assoc _ _ _)
  COALGEBRA .isSetHom {x}{y} =
    isSetΣ (C .isSetHom)
      (λ ϕ → isProp→isSet (isPropIsCoalgebraHom (x .snd) (y .snd) ϕ))
