module CYK {Σ' : Set 0ℓ} {G : Grammar Σ'}
  (_≟_ : DecidableEquality Σ')
  where

open Grammar G
open Context-Free-CNF-Grammar.Semantics Σ' G

ℓ  : Level
ℓ  = 0ℓ

-- Our index set:
I : Set ℓ
I = List⁺ Σ'

-- Our base category
-- The level is chosen to be suitable high for the algebra A later
𝒞 : Set _
𝒞 = Set ℓ

-- The coalgebra carrier:
C : I → 𝒞
C w = Lift _ ⊤

-- The algebra carrier sends w to a list containing all symbols V which produce w:
A : I → 𝒞
A w = AllInhabitants (λ V → V -+→ w)

-- The functor of interest:
F₀ : (I → 𝒞) → (I → 𝒞)
F₀ X w =
  -- We write the dependent product as function for simplicity
  (u v : List⁺ Σ') → u ⁺++⁺ v ≡ w → X u × X v

-- The notation ⇒ refers to a morphism in the power category  𝒞 ^ I
F₁ : {X Y : I → 𝒞} → (X ⇒ Y) → F₀ X ⇒ F₀ Y
F₁ {X} {Y} f w p u v u++v≡w = f u (proj₁ Xu×Xv) , f v (proj₂ Xu×Xv)
  where
    Xu×Xv : X u × X v
    Xu×Xv = p u v u++v≡w

-- The coalgebra structure:
coalg : C ⇒ F₀ C
coalg w (Level.lift tt) u v uv≡w = (Level.lift tt) , (Level.lift tt)

-- The algebra structure:
alg : F₀ A ⇒ A
-- Case 1: words of length 1
alg (σ ∷ []) _ =
  let
    -- it is decidable, whether σ is in one of
    -- the production rules for P:
    σ-in-rules? : Unary.Decidable (λ P → σ ∈ (R-T P))
    σ-in-rules? = (λ P → Dec._∈?_ _≟_ σ (R-T P))
    -- since the grammar is in Chomsky normal form,
    -- this is equivalent to the equestion whether
    -- P can produces the word [ σ ] at all, and therefore
    -- yields a decision procedure for producability:
    produces-σ? : Unary.Decidable (λ P → P -+→ [ σ ])
    produces-σ? = Unary.map (-+→-T , -+→-T⁻¹) σ-in-rules?
  in
  -- so we use this decision procedure to filter all (non-terminal) symbols from
  -- V by the property whether P can produce σ:
  mkAllInhabitantsByFilter _ allV allV∈ produces-σ? 

-- Case 2: words w of length ≥ 2
alg w@(_ ∷ _ ∷ _) p = all-P-with-P→w
  where
    -- For symbols Q, T a predicate telling whether a symbol P can produce Q T
    ⟶V : V × V → V → Set 0ℓ
    ⟶V Q,T = λ P → Q,T ∈ R-V P

    ⟶V? : ∀ Q,T → Unary.Decidable (⟶V Q,T)
    ⟶V? Q,T P = Dec._∈?_ (Product.≡-dec Fin._≟_ Fin._≟_) Q,T (R-V P)

    open do-AllInhabitants -- used in below do-notation

    -- we construct a more explicit list
    -- of all tuples (u,v,Q,T,P) that witness P -+→ w
    computation : AllInhabitants
               {A = (List⁺ Σ' × List⁺ Σ') × V × V × V}
               (λ { ((u , v) , Q , T , P) →
                (u ⁺++⁺ v ≡ w) × (Q -+→ u) × (T -+→ v) × ((Q , T) ∈ R-V P)})
    computation = do
      ((u , v) , uv≡w) ← all-decompositions w
      let puv = (A u × A v) ∋ p u v uv≡w
      (Q , Q→u) ← proj₁ puv -- Some Q that produces u
      (T , T→v) ← proj₂ puv -- Some T that produces v
      mkAllInhabitantsByFilter (⟶V (Q , T)) allV allV∈ (⟶V? (Q , T))

    -- We now map the redundant information away:
    all-P-with-P→w : AllInhabitants (λ P → P -+→ w)
    all-P-with-P→w =
      map-AllInhabitants (proj₂ ∘ proj₂ ∘ proj₂)
      (λ { ((u , v) , Q , T , P)
           (uv≡w , Q→u , T→v , P→QT)
           →
           subst (P -+→_) uv≡w (-+→-V P→QT Q→u T→v)
      })
      (λ P P→w →
        let
          (Q + u ⊣ Q→u) , (T + v ⊣ T→v) , P→QT , uv≡w =  -+→-V⁻¹ P→w
        in
        ((u , v) , (Q , T , P)) , (refl , uv≡w , (Q→u , T→v , P→QT))
      )
      computation

-- we order words by length:
_≺_ : I → I → Set _
_≺_ = ℕ._<_ on length

≺-is-well-founded : WellFounded _≺_
≺-is-well-founded = wellFounded length  <-wellFounded

open import IntrinsicallyRecursiveCoalgs _≺_ using (_↓₀; module c2a)
-- the functor in use is well-founded:
Fwf : {X : List⁺ Σ' → 𝒞} → F₀ X ⇒ (F₀ ↓₀) X
Fwf {X} w p u v u++v≡w = (u≺w , proj₁ (p u v u++v≡w)) , v≺w , proj₂ (p u v u++v≡w)
  where
  |u|+|v|≡|w| : length u ℕ.+ length v ≡ length w
  |u|+|v|≡|w| = (subst (λ - → _ ≡ length -) u++v≡w (sym (List⁺.length-⁺++⁺ u v)))

  u≺w : u ≺ w
  u≺w = subst (λ - → length u ℕ.< -) (|u|+|v|≡|w|) (ℕ.m<m+n (length u) (ℕ.s≤s ℕ.z≤n))
  v≺w : v ≺ w
  v≺w = subst (λ - → length v ℕ.< -) (trans (ℕ.+-comm (length v) (length u)) |u|+|v|≡|w|) (ℕ.m<m+n (length v) (ℕ.s≤s ℕ.z≤n))


CYK-c2a : C ⇒ A
CYK-c2a = c2a.c2a-morph ≺-is-well-founded F₀ F₁ Fwf coalg alg

-- CYK sends each word w to the list of all symbols P that produce w:
CYK : ∀ (w : List⁺ Σ') → AllInhabitants (λ P → P -+→ w)
CYK w = CYK-c2a w (Level.lift tt)
