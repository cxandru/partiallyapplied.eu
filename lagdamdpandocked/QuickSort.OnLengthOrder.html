private
  variable
    ℓ : Level
    A B C D : Type ℓ

infixl 1 _on_

-- Taken from `Relation.Binary.Construct.On` from the stdlib
-- (Can't use it directly because `Acc` is the cubical one)

_on_ : (B → B → C) → (D → B) → (D → D → C)
_*_ on f = \x y -> f x * f y

module _ (f : B → D) where
  accessible : ∀ {∼ : D → D → Type ℓ} {x} → Acc ∼ (f x) → Acc (∼ on f) x
  accessible (acc rs) = acc (λ y fy<fx → accessible (rs (f y) fy<fx))

  wellFounded : {∼ : D → D → Type ℓ} → WellFounded ∼ → WellFounded (∼ on f)
  wellFounded wf x = accessible (wf (f x))


module _ {A : Type ℓ} where
  open Length {A = A}
  infix 4 _<l_
  opaque
    _<l_ : FMSet A → FMSet A → Type
    _<l_ = _<_ on length

    <l-Dec : (i j : FMSet A) → Dec (i <l j)
    <l-Dec i j = <Dec (length i) (length j)

    <l-∷ : ∀ {x xs} → length xs < length (x ∷ xs)
    <l-∷ {x} {xs} = 0 , sym length-∷

    open import Cubical.Foundations.HLevels
    <l-Prop : (a b : FMSet A) → isProp (a <l b)
    <l-Prop a b = isProp≤

    <l-wellfounded : WellFounded _<l_
    <l-wellfounded = wellFounded length <-wellfounded
