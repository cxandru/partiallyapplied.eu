private variable
  ℓ : Level

-- The carrier C of the coalgebra is the singleton type for `(ℕ × ℕ)`
-- So in terms of slice category, C is ((ℕ × ℕ), id) ∈ Set / (ℕ × ℕ)
C : ℕ × ℕ → Set
C _ = ⊤

-- Sending two numbers to the predicate of their common divisors:
common-divisor : ℕ → ℕ → (ℕ → Set)
common-divisor k l q = (q ∣ k) × (q ∣ l) 

-- greatest (w.r.t. _∣_) natural number satisfying a predicate P
greatest : (ℕ → Set) → Set
greatest P = Σ[ g ∈ ℕ ] P g × ∀ d → P d → d ∣ g

A : ℕ × ℕ → Set
A (m , n) = greatest (common-divisor m n)
-- In set notation: A (m , n) = { g ∈ ℕ | g divides m and n and is the greatest such }

-- The observation: any Set^I-morphism C ⇒ A yields a correct GCD implementation:
observation : ∀ (h : C ⇒ A) → ∀ m n → greatest (common-divisor m n)
observation h m n = h (m , n) tt

-- The base functor G for the recursion expresses that we calculate
-- `gcd(a , 0)` from `a`, and `gcd(b + q * a , a)` from `(a , b)`
data G (X : (ℕ × ℕ) → Set ℓ) : (ℕ × ℕ) → Set ℓ where
  -- intuitively the base case
  Inl : (a : ℕ) → G X (a , 0)
  -- Corresponding to the recursive call
  Inr : {a b : ℕ} → (q : ℕ) → X (a , b) → a > b → G X (b + q * a , a)

-- When writing this functor on a slice category, that is, for a set X with a
-- function
--     ⟨r₁,r₂⟩ : X → ℕ × ℕ
-- then above defintion of G corresponds to the following functor:
--     G (X,⟨r₁,r₂⟩) = ℕ + ℕ × { x ∈ X ∣ r₁(x) > r₂(x) }
--     r_G : G (X,⟨r₁,r₂⟩) → Set
--     r_G(inl(a)) = (a , 0)
--     r_G(inr(q,x)) = (r₂(x) + q * r₁(x) , r₁(x))

-- The action of the functor on morphisms in Set ^ (ℕ × ℕ)
G₁ : {X Y : ℕ × ℕ → Set ℓ} →
 (X ⇒ Y) → (G X ⇒ G Y)
G₁ _ _ (Inl a) = Inl a
G₁ f _ (Inr q x a>b) = Inr q (f (_ , _) x) a>b

-- The ordering on the index w.r.t. which `GCDS` is well founded
private
  _<₂_ : Rel (ℕ × ℕ) 0ℓ
  _<₂_ = _<_ on proj₂

  <₂-wellFounded : WellFounded _<₂_
  <₂-wellFounded = InductOn.wellFounded proj₂ <-wellFounded

open import IntrinsicallyRecursiveCoalgs (_<₂_) as U

-- The inverse to the wellfounded coreflection counit witnessing the
-- wellfoundedness of G
G-ε⁻¹ : {X : (ℕ × ℕ) → Set ℓ} → (o : ℕ × ℕ) → G X o → (G ↓₀) X  o
G-ε⁻¹ _ (Inl a) = Inl a
G-ε⁻¹ _ (Inr q x a>b) = Inr q (a>b , x) a>b

coalg : C ⇒ G C
coalg (a , zero) tt = Inl a
-- If b is non-zero, then using `a divMod b`, we can rewrite a ≡ q * b + r:
coalg (a , b@(suc _)) tt with a divMod b
... | result q r refl = Inr
          -- Parameters for recursive call:
          {a = b} {b = toℕ r}
          -- Additional data for the rank:
          q tt (toℕ<n r)

alg : G A ⇒ A
alg .(a , 0) (Inl a) =
  a  -- a is a common divisor of a and 0:
  , (n∣n , a ∣0)
  -- any other common divisor d also divides a:
  , λ d (d∣a , _) → d∣a
alg .(b + q * a , a) (Inr {a} {b} q (g , (g∣a , g∣b) , g-is-greatest) _) =
  -- if g is the greatest common divisor of a and b, then it is
  -- also the greatest common divisor of b + q * a and a:
  g
  -- 1. g divides both b + q * a and a:
  , (∣m∣n⇒∣m+n g∣b (∣n⇒∣m*n q g∣a) , g∣a)
  -- 2. g is the greatest such:
  , λ d (d∣b+q*a , d∣a) →
    -- if d divides both b + q * a and a, then d also divides b
    -- also divides g since g was the gcd of a and b:
    g-is-greatest _ (d∣a
                    , ((d ∣ b) ∋ (∣m+n∣m⇒∣n (subst (d ∣_) (+-comm b _ ) d∣b+q*a) (∣n⇒∣m*n q d∣a))))

-- Instantiation of our framework with the proofs that the functor G is well-founded
GCD-coalg→alg : C ⇒ A
GCD-coalg→alg = U.c2a.c2a-morph <₂-wellFounded G G₁ G-ε⁻¹ coalg alg

gcd : ∀ k l → greatest (common-divisor k l)
gcd = observation GCD-coalg→alg

-- Example terms to demonstrate that our definition computes
_ : gcd 96 15 ≡ (3 , (divides-refl 32 , divides-refl 5) , _)
_ = refl
_ : gcd 127 128 ≡ (1 , (divides-refl 127 , divides-refl 128) , _)
_ = refl
_ : gcd 12 20 ≡ (4 , (divides-refl 3 , divides-refl 5) , _)
_ = refl
_ : gcd 0 36 ≡ (36 , (divides-refl 0 , divides-refl 1) , _)
_ = refl
_ : gcd 17 0 ≡ (17 , (divides-refl 1 , divides-refl 0) , _)
_ = refl
