module Related-Definitions {a b ℓ : Level} {A : Set a} {B : Set b} where
  record Related (R : A → B → Set ℓ) : Set (a ⊔ b ⊔ ℓ) where
    constructor _+_⊣_
    field
      fst : A
      snd : B
      rel : R fst snd

    ₁ = fst
    ₂ = snd

    to-∃₂ : ∃[ x ] ∃[ y ] R x y
    to-∃₂ = fst , snd , rel

open Related-Definitions public

module pull-P-Definitions {a b : Level} {A : Set a} where
  pull-P-in-List : {P : A → Set b} → (l : List A) → (∀ {x} → x ∈ l → P x) → List (Σ A P)
  pull-P-in-List [] allP = []
  pull-P-in-List (x ∷ l) allP = (x , allP (here refl)) ∷ pull-P-in-List l λ x∈l → allP (there x∈l)
  
  proj₁∘pull-P-in-List : {P : A → Set b} → (l : List A) → (allP : ∀ {x} → x ∈ l → P x)
                            → List.map proj₁ (pull-P-in-List l allP) ≡ l
  proj₁∘pull-P-in-List [] allP = refl
  proj₁∘pull-P-in-List (x ∷ l) allP =
    cong (x ∷_) (proj₁∘pull-P-in-List l λ x∈l → allP (there x∈l))

  filter' : {P : A → Set b} → (Unary.Decidable P) → List A → List (Σ A P)
  filter' P? [] = []
  filter' P? (x ∷ xs)  with (P? x)
  ... | (yes bc) = (x , bc) ∷ (filter' P? xs)
  ... | (no _) = filter' P? xs

  proj₁∘filter' : {P : A → Set b} → (P? : Unary.Decidable P) → (xs : List A)
                → List.map proj₁ (filter' P? xs) ≡ filter P? xs
  proj₁∘filter' P? [] = refl
  proj₁∘filter' P? (x ∷ xs) with (P? x)
  ... | (yes bc) = cong (x List.∷_) (proj₁∘filter' P? xs)
  ... | (no _) = (proj₁∘filter' P? xs)


  elim-map₂ : {c : Level} {B : A → Set b} {C : A → Set c}
              → (l : List (Σ A B))
              → (f : ∀ {x} → B x → C x)
              → List.map proj₁ (List.map (map₂ f) l) ≡ List.map proj₁ l
  elim-map₂ {B = B} l f =
    begin
    List.map proj₁ (List.map (map₂ f) l) ≡⟨ List.map-∘ l ⟨
    List.map (proj₁ ∘ map₂ f) l ≡⟨ List.map-cong pm≡p l ⟩
    List.map proj₁ l
    ∎
    where
      pm≡p : ∀ (x : Σ A B) → (proj₁ ∘ map₂ f) x ≡ proj₁ x
      pm≡p (fst , snd) = refl
      open ≡.≡-Reasoning 

open pull-P-Definitions public

module with-f {a b : Level} {A : Set a} {B : Set b}
       (f : A → List B)
       {y : B}
       {xs : List A}
    where
  ∈∈-concatMap⁺ :  ∀ {x : A} → x ∈ xs → y ∈ f x → y ∈ List.concatMap f xs
  ∈∈-concatMap⁺ {x} x∈xs y∈fx =
    ∈-concat⁺′ y∈fx fx∈f[xs]
    where
      fx∈f[xs] : f x ∈ List.map f xs
      fx∈f[xs] = ∈-map⁺ f {xs = xs} x∈xs

  ∈∈-concatMap⁻ :  y ∈ List.concatMap f xs → Σ[ x ∈ A ] y ∈ f x × x ∈ xs
  ∈∈-concatMap⁻ y∈f*x =
    let
      ys , y∈ys , ys∈f[x] = ∈-concat⁻′ (List.map f xs) y∈f*x
      x , x∈xs , ys≡fx = ∈-map⁻ _ ys∈f[x]
    in
    x , subst (y ∈_) ys≡fx y∈ys , x∈xs

open with-f public
