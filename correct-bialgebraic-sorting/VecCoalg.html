{-# OPTIONS --cubical --exact-split #-}
module VecCoalg where
open import Cubical.Foundations.Prelude
open import Cubical.Foundations.Function
open import Cubical.Data.Vec hiding (replicate ; length)
open import Cubical.Data.Nat as Nat
open import Cubical.Data.Sigma

private
  variable
    ℓ ℓ' ℓ'' : Level
    A : Type ℓ
    B : Type ℓ'

data V (A : Type ℓ) (r : ℕ → Type ℓ') : ℕ → Type (ℓ-max ℓ ℓ') where
  [] : V A r zero
  _∷_ : ∀ {n} (x : A) → r n → V A r (suc n)

V′ : (A : Type ℓ) (r : ℕ → Type ℓ') → ℕ → Type (ℓ-max ℓ ℓ')
V′ A r zero = Unit*
V′ A r (suc n) = A × r n

〖_〗 : {A : Type ℓ} → {r : ℕ → Type ℓ'} → ({nₓ : ℕ} → r nₓ → V A r nₓ) →
  {n : ℕ} → r n → Vec A n
〖 grow 〗 seed with grow seed
...| [] = []
...| x ∷ seed′ = x ∷ 〖 grow 〗 seed′

〖_〗′ : {A : Type ℓ} → {r : ℕ → Type ℓ'} → ({nₓ : ℕ} → r nₓ → V′ A r nₓ) →
  {n : ℕ} → r n → Vec A n
〖_〗′ {A = A} {r = r} grow {n = n} = Nat.elim {A = λ m → r m → Vec A m} (λ _ → []) bar n
  where
   bar : ∀ n₁ → (r n₁ → Vec A n₁) → r (suc n₁) → Vec A (suc n₁)
   bar m IH seed with grow seed
   ...| x , seed′ = x ∷ IH seed′


data Singleton {ℓ : _} {A : Type ℓ} : A -> Type ℓ where
  singleton : (a : A) -> Singleton a

-- 😂 this is a most abstruse definition
data M (r : ℕ → Type ℓ) : ℕ → Type ℓ where
  zero : M r zero
  suc : ∀ {n} → r n → M r (suc n)

〖_〗M : {r : ℕ → Type ℓ'} → ({nₓ : ℕ} → r nₓ → M r nₓ) →
  {n : ℕ} → r n → Singleton n
〖_〗M grow seed with grow seed
...| zero = singleton zero
...| suc seed′ with 〖_〗M grow seed′
...| singleton m = singleton $ suc m

replicate : A → (n : ℕ) → Vec A n
replicate a n = 〖_〗 {r = Singleton} (foo a) (singleton n)
  where
    foo : A → ∀ {nₓ} → Singleton nₓ → V A Singleton nₓ
    foo a {nₓ = zero} _ = []
    foo a {nₓ = suc nₓ} _ = a ∷ singleton nₓ

length : {n : ℕ} → Vec A n → Singleton n
length {A = A} = 〖_〗M {r = Vec A} foo
  where
    foo : ∀ {nₓ : ℕ} → Vec A nₓ → M (Vec A) nₓ
    foo [] = zero
    foo (x ∷ l) = suc l
