module VecCoalg where
open import Level hiding (zero ; suc)
open import Data.Vec hiding (replicate ; length)
open import Data.Nat as Nat hiding (_⊔_)

private
  variable
    ℓ ℓ' : Level
    A : Set ℓ

ℕ-elim : ∀ {ℓ} {P : ℕ → Set ℓ}
  → P zero
  → ((n : ℕ) → P n → P (suc n))
  → (n : ℕ) → P n
ℕ-elim a₀ _ zero = a₀
ℕ-elim a₀ f (suc n) = f n (ℕ-elim a₀ f n)

-- `Vec` Base functor 
data V (A : Set ℓ) (r : ℕ → Set ℓ') : ℕ → Set (ℓ ⊔ ℓ') where
  [] : V A r zero
  _∷_ : ∀ {n} (x : A) → r n → V A r (suc n)

-- Anamorphism / morphism into the final coalgebra – `Vec` is the final `V` coalgebra, where `V` : Set^ℕ.
-- This definition is accepted because the implicit argument `{n : ℕ}` decreases in the recursive call.
-- This because in the case `n ≡ suc n₁` `seed′` is of type `r n₁`.
〖_〗 : {A : Set ℓ} → {r : ℕ → Set ℓ'} →
  ({nₓ : ℕ} → r nₓ → V A r nₓ) →
  {n : ℕ} → r n → Vec A n
〖 grow 〗 seed with grow seed
...| [] = []
...| x ∷ seed′ = x ∷ 〖 grow 〗 seed′

-- Here we make the passing of the implicit decreasing argument explicit
ana : {A : Set ℓ} → {r : ℕ → Set ℓ'} →
  ({nₓ : ℕ} → r nₓ → V A r nₓ) →
  {n : ℕ} → r n → Vec A n
ana coalg {n} x₀ with coalg x₀
ana coalg .{0} x₀      | []                   = []
ana coalg .{suc n′} x₀ | _∷_ {n = n′} x seed′ = x ∷ ana coalg {n′} seed′


-- We can make this explicit by defining the map as a family of morphisms `λ m → r m → Vec A m` indexed by `n`.
-- We use the dependent elimination principle `elim` to inductively define this family of morphisms.
〖_〗′ : {A : Set ℓ} → {r : ℕ → Set ℓ'} →
  ({nₓ : ℕ} → r nₓ → V A r nₓ) →
  {n : ℕ} → r n → Vec A n
〖_〗′ {A = A} {r = r} grow {n = n} = ℕ-elim
  {P = λ m → r m → Vec A m}
  (λ _ → []) IS n
  where
   IS : ∀ n₁ → (r n₁ → Vec A n₁) → r (suc n₁) → Vec A (suc n₁)
   IS m IH seed with grow seed
   ...| x ∷ seed′ = x ∷ IH seed′
